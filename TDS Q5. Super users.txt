5. Super users
A company defines its super users as those who have made at least two transactions. 
From the following table, write a query to return, for each user, 
the date when they become a super user, ordered by oldest super users first. Users 
who are not super users should also be present in the table.

select user_id, super_user_date 
FROM
(
select user_id, transaction_date as super_user_date from
(
(select user_id, transaction_date, dense_rank()OVER (partition by user_id order by transaction_date) as rnk
from users) t1
where rnk =2
) t1

UNION

select user_id, NULL as super_user_date from    --- NOTE THIS CAN BE DONE!

having count(*) <2

)t2
order by super_user_date

df['rank']=df.sort_values(by=['user_id', 'transaction_date'], ascending =[True , True]).groupby('user_id')['transaction_date'].rank(method='dense', ascending= True).asType(int)

super_user=df.loc[df['rank']==2, ['user_id', 'transaction_date']]
df_grouped=df.groupby(['user_id'], as_index=False).agg(count_transactions =('user_id', 'count))
